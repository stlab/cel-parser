cmake_minimum_required(VERSION 3.24)
project(installed_parser_consumer LANGUAGES CXX)
find_package(cel-parser CONFIG REQUIRED)
get_target_property(parser_imported cel::parser IMPORTED)
if(NOT parser_imported)
    message(FATAL_ERROR "The consumer requires an installed imported target")
endif()
include(CTest)
add_executable(consumer ../../examples/example.cpp)
target_link_libraries(consumer PRIVATE cel::parser)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(consumer PRIVATE /W4 /WX)
else()
    target_compile_options(consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME consumer COMMAND consumer)
add_executable(lookup_consumer lookup.cpp)
target_link_libraries(lookup_consumer PRIVATE cel::parser)
set_target_properties(lookup_consumer PROPERTIES CXX_EXTENSIONS OFF)
add_test(NAME lookup_consumer COMMAND lookup_consumer)
add_executable(compiler_consumer compiler.cpp)
target_link_libraries(compiler_consumer PRIVATE cel::parser)
set_target_properties(compiler_consumer PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(compiler_consumer PRIVATE /W4 /WX)
else()
    target_compile_options(compiler_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME compiler_consumer COMMAND compiler_consumer)
add_executable(cached_consumer ../../examples/cached_decoded.cpp)
target_link_libraries(cached_consumer PRIVATE cel::parser)
set_target_properties(cached_consumer PROPERTIES CXX_EXTENSIONS OFF)
add_test(NAME cached_consumer COMMAND cached_consumer)
add_executable(measurement_consumer ../../tools/measurement.cpp)
target_link_libraries(measurement_consumer PRIVATE cel::parser)
set_target_properties(measurement_consumer PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_CLANG_TIDY)
    # Match the bounded impossible-array-path exclusion on the measurement target.
    set_property(TARGET measurement_consumer APPEND PROPERTY CXX_CLANG_TIDY
        "--checks=-clang-analyzer-unix.cstring.NullArg")
endif()
add_test(NAME measurement_consumer COMMAND measurement_consumer 10)
foreach(client lookup_consumer cached_consumer measurement_consumer)
    if(MSVC)
        target_compile_options(${client} PRIVATE /W4 /WX)
    else()
        target_compile_options(${client} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/imported-parser-$<CONFIG>.txt"
    CONTENT "library=$<TARGET_FILE:cel::parser>\nheaders=$<TARGET_PROPERTY:cel::parser,INTERFACE_INCLUDE_DIRECTORIES>\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/imported-runtime-$<CONFIG>.txt"
    CONTENT "library=$<TARGET_FILE:cel::runtime>\nheaders=$<TARGET_PROPERTY:cel::runtime,INTERFACE_INCLUDE_DIRECTORIES>\n")
